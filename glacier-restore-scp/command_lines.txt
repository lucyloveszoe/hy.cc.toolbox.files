================== Glacier Restore 

python glacier_restore.py <bucket> <prefix>
    [--days DAYS]                       # default 7
    [--tier {Bulk,Standard,Expedited}]  # default Bulk
    [-x PATTERN]...                     # replaces default excludes
    [--add-exclude PATTERN]...          # added to defaults
    [--profile PROFILE] [--region REGION]
    [--execute]
    
python glacier_restore.py yu.vbackup.s  records/2018.06.18/report_mcjt/   --days 2 --execute --add-exclude "Thumbs.db" --add-exclude ".DS_Store"

    [--profile PROFILE] [--region REGION]
    [--dry-run]
